    async function parseApiResponse(response) {
      let body = null;
      const contentType = response.headers.get("content-type") || "";
      if (contentType.includes("application/json") || contentType.includes("application/problem+json")) {
        body = await response.json();
      } else {
        const text = await response.text();
        try {
          body = text ? JSON.parse(text) : null;
        } catch (error) {
          body = text;
        }
      }

      if (!response.ok) {
        throw makeGitHubError(body, response.status, {
          "x-ratelimit-remaining": response.headers.get("x-ratelimit-remaining"),
          "x-ratelimit-reset": response.headers.get("x-ratelimit-reset"),
          "retry-after": response.headers.get("retry-after")
        });
      }

      return { body, response };
    }

    async function apiFetchCors(path, options = {}, includeToken = true) {
      const method = String(options.method || "GET").toUpperCase();
      const suppliedHeaders = { ...(options.headers || {}) };
      if (!includeToken) {
        delete suppliedHeaders.Authorization;
        delete suppliedHeaders.authorization;
      }
      const headers = {
        ...apiHeaders(includeToken),
        ...suppliedHeaders
      };
      const requestOptions = {
        ...options,
        method,
        mode: "cors",
        credentials: "omit",
        cache: "no-store",
        signal: options.signal || (state.requestController ? state.requestController.signal : undefined)
      };
      if (Object.keys(headers).length) {
        requestOptions.headers = headers;
      } else {
        delete requestOptions.headers;
      }
      const response = await fetch(`${API_ROOT}${path}`, requestOptions);
      return parseApiResponse(response);
    }

    function apiFetchXhr(path, options = {}, includeToken = false) {
      return new Promise((resolve, reject) => {
        const method = String(options.method || "GET").toUpperCase();
        if (method !== "GET") {
          const error = new Error("XHR compatibility mode only supports GET requests.");
          error.transport = "xhr";
          reject(error);
          return;
        }

        const xhr = new XMLHttpRequest();
        const signal = options.signal || (state.requestController ? state.requestController.signal : null);
        let settled = false;

        const cleanup = () => {
          if (signal) signal.removeEventListener("abort", onAbort);
        };
        const fail = error => {
          if (settled) return;
          settled = true;
          cleanup();
          reject(error);
        };
        const onAbort = () => {
          try { xhr.abort(); } catch (error) { /* no-op */ }
          fail(new DOMException("The request was aborted.", "AbortError"));
        };

        xhr.open("GET", `${API_ROOT}${path}`, true);
        xhr.responseType = "text";
        xhr.timeout = 20000;
        xhr.withCredentials = false;
        if (includeToken && state.token) {
          xhr.setRequestHeader("Accept", "application/vnd.github+json");
          xhr.setRequestHeader("Authorization", `Bearer ${state.token}`);
        }

        xhr.onload = () => {
          if (s
